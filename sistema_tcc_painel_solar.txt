function Escrever($caminho, $texto) {
    $dir = Split-Path $caminho -Parent
    if (-not (Test-Path $dir)) { New-Item -ItemType Directory -Path $dir -Force | Out-Null }
    [System.IO.File]::WriteAllText($caminho, $texto, (New-Object System.Text.UTF8Encoding $false))
}

function Achar-Python {
    foreach ($cmd in @('python', 'py')) {
        try {
            $v = & $cmd --version 2>&1 | Out-String
            if ($LASTEXITCODE -eq 0 -and $v -match 'Python 3\.(\d+)') {
                if ([int]$Matches[1] -ge 10) { return $cmd }
            }
        } catch {}
    }
    return $null
}

#include <Arduino.h>

// =====================================================
// PINOS
// =====================================================

// Sensor infravermelho
const uint8_t SENSOR_PIN = A0;

// Relés
const uint8_t RELE_BOMBA_1 = 7;  // IN1 / K1
const uint8_t RELE_BOMBA_2 = 8;  // IN2 / K2


// =====================================================
// CONFIGURAÇÃO DO RELÉ
// =====================================================

// A maioria dos módulos de relé para Arduino é ACTIVE LOW:
//
// LOW  = relé ligado
// HIGH = relé desligado
//
// Se o seu funcionar ao contrário, troque os dois valores.

const uint8_t RELE_LIGADO = LOW;
const uint8_t RELE_DESLIGADO = HIGH;


// =====================================================
// CONFIGURAÇÃO DO SENSOR
// =====================================================

// Se aumentar a sujeira fizer a leitura do A0 DIMINUIR:
// deixe false.
//
// Se aumentar a sujeira fizer a leitura do A0 AUMENTAR:
// coloque true.

const bool SENSOR_INVERTED = false;


// =====================================================
// LIMITE DE SUJIDADE
// =====================================================

// Painel limpo = 100%
//
// O sistema será acionado quando o sinal cair 5%.
//
// 100% - 5% = 95%

const float TRIGGER_PERCENT = 95.0;


// =====================================================
// CONFIRMAÇÃO
// =====================================================

// Número de leituras seguidas abaixo de 95% antes de
// considerar que realmente existe sujeira.

const uint8_t CONFIRMATION_READS = 8;


// =====================================================
// TEMPOS DAS BOMBAS
// =====================================================

// Tempo que as bombas ficarão ligadas.
//
// 5000 ms = 5 segundos.
//
// PARA O PRIMEIRO TESTE:
// deixe apenas 5 segundos.

const unsigned long TEMPO_BOMBAS = 5000;


// Depois da limpeza espera 10 segundos antes
// de começar uma nova análise.

const unsigned long TEMPO_ESPERA_APOS_LIMPEZA = 10000;


// =====================================================
// LEITURAS
// =====================================================

const uint16_t CALIBRATION_SAMPLES = 150;
const uint8_t READ_SAMPLES = 25;


// =====================================================
// VARIÁVEIS
// =====================================================

float baselineSignal = 0.0;

uint8_t contadorSujeira = 0;


// =====================================================
// FUNÇÃO: MÉDIA DO SENSOR
// =====================================================

float lerSensorMedia(uint16_t quantidade) {

    unsigned long soma = 0;

    for (uint16_t i = 0; i < quantidade; i++) {

        soma += analogRead(SENSOR_PIN);

        delay(5);
    }

    return (float)soma / quantidade;
}


// =====================================================
// FUNÇÃO: CORRIGIR SENTIDO DO SENSOR
// =====================================================

float converterSinal(float leitura) {

    if (SENSOR_INVERTED) {

        return 1023.0 - leitura;
    }

    return leitura;
}


// =====================================================
// FUNÇÃO: DESLIGAR BOMBAS
// =====================================================

void desligarBombas() {

    digitalWrite(RELE_BOMBA_1, RELE_DESLIGADO);

    digitalWrite(RELE_BOMBA_2, RELE_DESLIGADO);
}


// =====================================================
// FUNÇÃO: LIGAR BOMBAS
// =====================================================

void ligarBombas() {

    digitalWrite(RELE_BOMBA_1, RELE_LIGADO);

    digitalWrite(RELE_BOMBA_2, RELE_LIGADO);
}


// =====================================================
// CICLO DE LIMPEZA
// =====================================================

void executarLimpeza() {

    Serial.println();
    Serial.println(F("======================================"));
    Serial.println(F("SUJIDADE DETECTADA"));
    Serial.println(F("INICIANDO LIMPEZA"));
    Serial.println(F("======================================"));


    // Liga os dois relés
    ligarBombas();


    Serial.println(F("K1 / BOMBA 1: LIGADA"));
    Serial.println(F("K2 / BOMBA 2: LIGADA"));


    // Mantém as bombas ligadas
    delay(TEMPO_BOMBAS);


    // Desliga
    desligarBombas();


    Serial.println();
    Serial.println(F("BOMBA 1: DESLIGADA"));
    Serial.println(F("BOMBA 2: DESLIGADA"));

    Serial.println(F("LIMPEZA FINALIZADA"));
    Serial.println();


    // Espera antes de voltar a medir
    Serial.println(F("Aguardando estabilizacao do sensor..."));

    delay(TEMPO_ESPERA_APOS_LIMPEZA);


    contadorSujeira = 0;
}


// =====================================================
// SETUP
// =====================================================

void setup() {

    Serial.begin(115200);


    // -----------------------------
    // Sensor
    // -----------------------------

    pinMode(SENSOR_PIN, INPUT);


    // -----------------------------
    // Relés
    // -----------------------------

    pinMode(RELE_BOMBA_1, OUTPUT);

    pinMode(RELE_BOMBA_2, OUTPUT);


    // IMPORTANTE:
    // Arduino sempre inicia com as bombas desligadas.

    desligarBombas();


    delay(2000);


    Serial.println();
    Serial.println(F("======================================"));
    Serial.println(F(" SISTEMA DE LIMPEZA DE PAINEL SOLAR"));
    Serial.println(F("======================================"));

    Serial.println();

    Serial.println(F("Sensor: A0"));

    Serial.println(F("Bomba 1: D7 / K1"));

    Serial.println(F("Bomba 2: D8 / K2"));

    Serial.println();


    // =================================================
    // CALIBRAÇÃO
    // =================================================

    Serial.println(F("CALIBRANDO SENSOR..."));

    Serial.println(F("Mantenha o sensor apontado para"));
    Serial.println(F("o painel LIMPO durante a calibracao."));

    Serial.println();


    delay(3000);


    float leituraInicial = lerSensorMedia(CALIBRATION_SAMPLES);


    baselineSignal = converterSinal(leituraInicial);


    Serial.println(F("CALIBRACAO CONCLUIDA"));

    Serial.print(F("Leitura A0: "));

    Serial.println(leituraInicial, 1);


    Serial.print(F("Referencia: "));

    Serial.print(baselineSignal, 1);

    Serial.println(F(" = 100%"));


    Serial.print(F("Limite de limpeza: "));

    Serial.print(TRIGGER_PERCENT);

    Serial.println(F("%"));


    Serial.println();

    Serial.println(F("--------------------------------------"));

    Serial.println(F("A0 | Sinal (%) | Queda (%) | Estado"));

    Serial.println(F("--------------------------------------"));
}


// =====================================================
// LOOP
// =====================================================

void loop() {

    // =================================================
    // LÊ SENSOR
    // =================================================

    float leituraBruta = lerSensorMedia(READ_SAMPLES);


    float sinal = converterSinal(leituraBruta);


    // =================================================
    // CALCULA PORCENTAGEM
    // =================================================

    float porcentagem = 100.0;


    if (baselineSignal > 0) {

        porcentagem =
            (sinal / baselineSignal) * 100.0;
    }


    float queda = 100.0 - porcentagem;


    if (queda < 0) {

        queda = 0;
    }


    // =================================================
    // MOSTRA NO SERIAL
    // =================================================

    Serial.print(leituraBruta, 1);

    Serial.print(F(" | "));

    Serial.print(porcentagem, 1);

    Serial.print(F("% | "));

    Serial.print(queda, 1);

    Serial.print(F("% | "));


    // =================================================
    // VERIFICA SUJEIRA
    // =================================================

    if (porcentagem <= TRIGGER_PERCENT) {

        contadorSujeira++;


        Serial.print(F("SUJEIRA "));

        Serial.print(contadorSujeira);

        Serial.print(F("/"));

        Serial.println(CONFIRMATION_READS);


        // =============================================
        // CONFIRMOU SUJEIRA
        // =============================================

        if (contadorSujeira >= CONFIRMATION_READS) {

            executarLimpeza();
        }
    }

    else {

        contadorSujeira = 0;

        Serial.println(F("NORMAL"));
    }


    delay(250);
}

# Sensor infravermelho de sujidade para painel solar

Projeto para Arduino Uno usando VS Code + PlatformIO.

## Critério de acionamento adotado

A versão inicial usava uma queda de 25% do sinal. Esse valor era arbitrário.

Nesta versão, o ponto inicial foi alterado para **5% de queda do sinal óptico**:

- referência com painel limpo = 100%
- acionamento = 95% ou menos
- recuperação/desligamento = 98% ou mais

O objetivo é detectar a sujeira antes que a perda de potência se torne grande.

### Fundamentação

**Alkharusi, T.; Alzahrani, M. M.; Pandey, C.; Yildizhan, H.; Markides, C. N. (2024).  
Experimental investigation of nonuniform PV soiling. Solar Energy, 272, 112493.  
DOI: 10.1016/j.solener.2024.112493**

O estudo experimental relata aproximadamente **6–7% de deterioração da potência para cada 5% de queda na transmitância**, nas condições estudadas.

Outro trabalho útil:

**Majeed, R.; Waqas, A.; Sami, H.; Ali, M.; Shahzad, N. (2020).  
Experimental investigation of soiling losses and a novel cost-effective cleaning system for PV modules. Solar Energy, 201, 298–306.  
DOI: 10.1016/j.solener.2020.03.014**

Nesse experimento, uma densidade de poeira de 4,6 g/m² esteve associada a queda de potência de aproximadamente **16,16% em módulo monocristalino** e **11,54% em policristalino** após um mês.

## Atenção científica importante

O artigo de Alkharusi et al. mede **transmitância óptica** do vidro/módulo.

O sensor conectado ao A0 deste projeto mede uma **resposta infravermelha/reflexiva**.

Por isso, não é correto afirmar automaticamente que:

`5% de queda no infravermelho = exatamente 5% de perda de potência`.

O limite de 95% deve ser tratado como um **ponto inicial de projeto baseado em literatura**.

Para transformar o sensor em um instrumento calibrado, o ideal é fazer um ensaio:

1. medir o A0 com o painel limpo;
2. medir tensão/corrente/potência do painel nas mesmas condições;
3. adicionar níveis controlados de poeira;
4. repetir as duas medições;
5. montar uma tabela relacionando queda do sensor e queda real de potência;
6. escolher o valor do A0 correspondente à perda máxima aceitável.

## Ligações

Sensor:

- VCC -> 5V
- GND -> GND
- AO -> A0

Saída:

- LED interno = indicação visual
- D7 = saída para futuro relé/MOSFET

**Não conecte uma bomba diretamente ao pino D7.**
Use relé ou MOSFET adequado e fonte apropriada para a bomba.

## Parâmetros que podem ser alterados

No `src/main.cpp`:

```cpp
const float TRIGGER_PERCENT = 95.0;
const float RELEASE_PERCENT = 98.0;
```

Exemplos:

- `95.0` = aciona com queda de 5%
- `90.0` = aciona com queda de 10%
- `85.0` = aciona com queda de 15%
- `75.0` = aciona com queda de 25%

## Sentido do sensor

```cpp
const bool SENSOR_INVERTED = false;
```

Se mais sujeira fizer a leitura de A0 diminuir, mantenha `false`.

Se mais sujeira fizer a leitura de A0 aumentar, use:

```cpp
const bool SENSOR_INVERTED = true;
```

import asyncio
import sqlite3
import threading
import time
from datetime import datetime
from pathlib import Path

import serial
from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles

PORTA = "COM3"      # porta USB do Arduino (NAO é a 8000)
BAUD = 115200
DB = "ativacoes.db"
PASTA_FRONT = Path(__file__).resolve().parent.parent / "front-end"

app = FastAPI()
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

clientes: set[WebSocket] = set()
diagnostico = {"linhas_recebidas": 0, "ultima_linha": "", "clientes_conectados": 0}


def iniciar_banco():
    with sqlite3.connect(DB) as c:
        c.execute("""CREATE TABLE IF NOT EXISTS ativacoes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            inicio TEXT, fim TEXT, duracao_seg INTEGER, queda_max REAL)""")


def salvar_ativacao(inicio, fim, queda_max):
    duracao = round((fim - inicio).total_seconds())
    with sqlite3.connect(DB) as c:
        c.execute("INSERT INTO ativacoes (inicio, fim, duracao_seg, queda_max) VALUES (?,?,?,?)",
                  (inicio.isoformat(), fim.isoformat(), duracao, queda_max))


async def transmitir(dados: dict):
    for ws in list(clientes):
        try:
            await ws.send_json(dados)
        except Exception:
            clientes.discard(ws)


def enviar(loop, dados):
    asyncio.run_coroutine_threadsafe(transmitir(dados), loop)


def ler_serial(loop):
    inicio_limpeza = None
    queda_max = 0.0
    ultima_queda = 0.0

    while True:
        try:
            with serial.Serial(PORTA, BAUD, timeout=1) as s:
                print(f"Conectado ao Arduino em {PORTA}")
                while True:
                    linha = s.readline().decode(errors="ignore").strip()
                    if not linha:
                        continue

                    print("ARDUINO:", linha)
                    diagnostico["linhas_recebidas"] += 1
                    diagnostico["ultima_linha"] = linha

                    if "INICIANDO LIMPEZA" in linha:
                        inicio_limpeza = datetime.now()
                        queda_max = ultima_queda
                        enviar(loop, {"estado": "limpando"})
                        continue

                    if "LIMPEZA FINALIZADA" in linha and inicio_limpeza:
                        salvar_ativacao(inicio_limpeza, datetime.now(), queda_max)
                        inicio_limpeza = None
                        enviar(loop, {"estado": "aguardando", "ativacao_salva": True})
                        continue

                    partes = [p.strip() for p in linha.split("|")]
                    if len(partes) == 4 and partes[1].endswith("%"):
                        try:
                            bruto = float(partes[0])
                            porcentagem = float(partes[1].rstrip("%"))
                            queda = float(partes[2].rstrip("%"))
                        except ValueError:
                            continue
                        ultima_queda = queda
                        queda_max = max(queda_max, queda)
                        estado = "sujeira" if partes[3].startswith("SUJEIRA") else "normal"
                        enviar(loop, {"bruto": bruto, "porcentagem": porcentagem,
                                      "queda": queda, "estado": estado})
        except serial.SerialException as e:
            print(f"Erro na serial: {e}. Tentando de novo em 3s...")
            time.sleep(3)


@app.on_event("startup")
async def startup():
    iniciar_banco()
    loop = asyncio.get_running_loop()
    threading.Thread(target=ler_serial, args=(loop,), daemon=True).start()


@app.websocket("/ws/sensor/")
async def ws_sensor(ws: WebSocket):
    await ws.accept()
    clientes.add(ws)
    print("Navegador conectado no WebSocket")
    try:
        while True:
            await ws.receive_text()
    except WebSocketDisconnect:
        clientes.discard(ws)


@app.get("/api/ativacoes/")
def listar_ativacoes(mes: str | None = None):
    mes = mes or datetime.now().strftime("%Y-%m")
    with sqlite3.connect(DB) as c:
        linhas = c.execute(
            "SELECT inicio, fim, duracao_seg, queda_max FROM ativacoes WHERE inicio LIKE ? ORDER BY inicio DESC",
            (f"{mes}%",)).fetchall()
    return [{"inicio": i, "fim": f, "duracao_seg": d, "queda_max": q} for i, f, d, q in linhas]


@app.get("/api/status/")
def status():
    diagnostico["clientes_conectados"] = len(clientes)
    return diagnostico


# SEMPRE por último: entrega o front-end em http://localhost:8010/
app.mount("/", StaticFiles(directory=PASTA_FRONT, html=True), name="front")

async function buscarAtivacoes() {
  if (CONFIG.USAR_SIMULACAO) {
    return Array.from({ length: 12 }, (_, i) => {
      const inicio = new Date(2026, 9, i * 2 + 1, 10, 0, 0);
      const duracao = 60 + Math.floor(Math.random() * 240);
      return { inicio, fim: new Date(inicio.getTime() + duracao * 1000), duracao_seg: duracao };
    });
  }
  const r = await fetch(`${CONFIG.API_URL}/ativacoes/`);
  const dados = await r.json();
  return dados.map(a => ({ ...a, inicio: new Date(a.inicio), fim: new Date(a.fim) }));
}

const graficoSinal = new Chart(document.getElementById("grafico-sinal"), {
  type: "line",
  data: {
    labels: [],
    datasets: [
      { label: "Sinal (%)", data: [], borderColor: "#f5a623", tension: 0.3 },

const TEXTOS = {
  normal: ["Normal", "normal"],
  sujeira: ["Sujeira detectada", "ativo"],
  limpando: ["LIMPANDO (bombas ligadas)", "ativo"],
  aguardando: ["Aguardando estabilizar", "normal"]
};

function mostrarEstado(estado) {
  const [texto, classe] = TEXTOS[estado] || ["--", ""];
  const el = document.getElementById("status");
  el.textContent = texto;
  el.className = classe;
}

function receberDado(d) {
  mostrarEstado(d.estado);

  if (d.porcentagem !== undefined) {
    const { labels, datasets } = graficoSinal.data;
    labels.push(new Date().toLocaleTimeString("pt-BR"));
    datasets[0].data.push(d.porcentagem);
    datasets[1].data.push(CONFIG.LIMITE);
    if (labels.length > CONFIG.MAX_PONTOS) {
      labels.shift();
      datasets[0].data.shift();
      datasets[1].data.shift();
    }
    graficoSinal.update();
  }

  // terminou uma limpeza: atualiza cards, tabela e gráfico do mês
  if (d.ativacao_salva) buscarAtivacoes().then(mostrarHistorico);
}

function iniciarTempoReal() {
  const ws = new WebSocket(CONFIG.WS_URL);
  ws.onopen = () => {
    document.getElementById("status").textContent = "Conectado, aguardando dados...";
  };
  ws.onmessage = (e) => receberDado(JSON.parse(e.data));
  ws.onerror = () => {
    document.getElementById("status").textContent = "Sem conexão com o servidor";
  };
  ws.onclose = () => setTimeout(iniciarTempoReal, 3000);
}

let graficoMensal = null;

function formatarDuracao(seg) {
  const m = Math.floor(seg / 60), s = seg % 60;
  return m ? `${m} min ${s} s` : `${s} s`;
}

// Escreve o número de ativações em cima de cada coluna
const rotulosNasColunas = {
  id: "rotulosNasColunas",
  afterDatasetsDraw(chart) {
    const { ctx } = chart;
    const meta = chart.getDatasetMeta(0);
    ctx.save();
    ctx.fillStyle = "#333";
    ctx.font = "bold 13px Arial";
    ctx.textAlign = "center";
    meta.data.forEach((barra, i) => {
      const valor = chart.data.datasets[0].data[i];
      if (valor > 0) ctx.fillText(valor, barra.x, barra.y - 6);
    });
    ctx.restore();
  }
};

function mostrarHistorico(ativacoes) {
  // ----- cards -----
  const total = ativacoes.reduce((soma, a) => soma + a.duracao_seg, 0);
  document.getElementById("total-ativacoes").textContent = ativacoes.length;
  document.getElementById("tempo-total").textContent = formatarDuracao(total);
  document.getElementById("tempo-medio").textContent =
    formatarDuracao(ativacoes.length ? Math.round(total / ativacoes.length) : 0);

  // ----- tabela -----
  document.querySelector("#tabela tbody").innerHTML = ativacoes.map(a => `
    <tr>
      <td>${new Date(a.inicio).toLocaleString("pt-BR")}</td>
      <td>${new Date(a.fim).toLocaleString("pt-BR")}</td>
      <td>${formatarDuracao(a.duracao_seg)}</td>
    </tr>`).join("");

  // ----- gráfico: uma coluna por dia do mês -----
  const hoje = new Date();
  const diasNoMes = new Date(hoje.getFullYear(), hoje.getMonth() + 1, 0).getDate();
  const contagem = Array(diasNoMes).fill(0);

  ativacoes.forEach(a => {
    const d = new Date(a.inicio);
    if (!isNaN(d)) contagem[d.getDate() - 1]++;
  });

  const dias = Array.from({ length: diasNoMes }, (_, i) => i + 1);
  const nomeMes = hoje.toLocaleDateString("pt-BR", { month: "long", year: "numeric" });

  if (graficoMensal) {
    graficoMensal.data.labels = dias;
    graficoMensal.data.datasets[0].data = contagem;
    graficoMensal.options.plugins.title.text = `Ativações da limpeza por dia: ${nomeMes}`;
    graficoMensal.update();
    return;
  }

  graficoMensal = new Chart(document.getElementById("grafico-mensal"), {
    type: "bar",
    data: {
      labels: dias,
      datasets: [{
        label: "Ativações",
        data: contagem,
        backgroundColor: "#ee5348",
        borderWidth: 0,
        barPercentage: 0.45,
        categoryPercentage: 1.0
      }]
    },
    options: {
      maintainAspectRatio: false,
      layout: { padding: { top: 20, right: 10 } },
      plugins: {
        legend: { display: false },
        title: {
          display: true,
          text: `Ativações da limpeza por dia: ${nomeMes}`,
          align: "start",
          color: "#333",
          font: { size: 18, weight: "normal" },
          padding: { bottom: 16 }
        },
        tooltip: {
          callbacks: {
            title: (itens) => `Dia ${itens[0].label}`,
            label: (item) => `${item.raw} ${item.raw === 1 ? "ativação" : "ativações"}`
          }
        }
      },
      scales: {
        x: {
          title: { display: true, text: "Dia do mês", color: "#555" },
          grid: { display: false },
          border: { color: "#bbb" }
        },
        y: {
          beginAtZero: true,
          title: { display: true, text: "(número de ativações)", color: "#555" },
          ticks: { stepSize: 1, precision: 0 },
          grid: { color: "#d9d9d9" },
          border: { color: "#bbb" }
        }
      }
    },
    plugins: [rotulosNasColunas]
  });
}
